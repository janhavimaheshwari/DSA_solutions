26.REMOVE DUPLIACATES FROM SORTED ARRAY
Solution:

class Solution(object):
    def removeDuplicates(self, nums):
        nlen= len(nums)

        if nlen == 0:
            return 0
            
        k=1
        
    
        for i in range(1,nlen):
        
            if nums[i]!=nums[i-1]:
                nums[k] = nums[i]
                k+=1

        return k
