121. BEST TIME TO BUY AND SELL STOCK

class Solution(object):
    def maxProfit(self, prices):
         maxProfit=0
         for i in range(len(prices)):
             for j in range(i+1,len(prices)):
                     profit = prices[j] - prices[i]
                
                     if profit > maxProfit:
                         maxProfit = profit

         return maxProfit

APPROACH- loops of i & j to point and compare 2 prices, and store (then compare again until maxProfit)

solution 2 - for better time complexity

class Solution(object):
    def maxProfit(self, prices):
        minPrice = float('inf')
        maxProfit = 0

        for price in prices:
            minPrice = min(minPrice, price)
            maxProfit = max(maxProfit, price - minPrice)

        return maxProfit
